<script>
const rows = document.querySelector("#ipTable").children;
const sound = document.getElementById("alertSound");
let lastStatus = Array.from(rows).map(() => 'offline');

function pingIP(ip, index) {
    // استخدام fetch لمحاولة الاتصال (CORS workaround)
    fetch("https://api.allorigins.win/get?url=" + encodeURIComponent("http://" + ip))
    .then(() => updateStatus(index, 'online'))
    .catch(() => updateStatus(index, 'offline'));
}

function updateStatus(index, status) {
    const td = rows[index].children[1];
    if(td.className !== status) {
        td.className = status;
        td.textContent = status.toUpperCase();
        if(lastStatus[index] !== status) sound.play();
        lastStatus[index] = status;
    }
}

function checkAll() {
    Array.from(rows).forEach((row, i) => {
        const ip = row.children[0].textContent;
        pingIP(ip, i);
    });
}

setInterval(checkAll, 5000); // التحديث كل 5 ثواني
checkAll();

// إضافة IP جديد
function addIP() {
    const ip = document.getElementById("ipInput").value.trim();
    if(!ip) return;
    const tbody = document.getElementById("ipTable");
    const tr = document.createElement("tr");
    const tdIP = document.createElement("td");
    tdIP.textContent = ip;
    const tdStatus = document.createElement("td");
    tdStatus.textContent = "Checking...";
    tdStatus.className = "offline";
    tr.appendChild(tdIP);
    tr.appendChild(tdStatus);
    tbody.appendChild(tr);
    lastStatus.push('offline');
    document.getElementById("ipInput").value = "";
}
</script>
